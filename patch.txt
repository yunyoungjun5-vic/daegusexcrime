const MainPopup = () => {
  const [isOpen, setIsOpen] = useState(false);

  useEffect(() => {
    const hideUntil = localStorage.getItem('hidePopupUntil');
    if (hideUntil && new Date().getTime() < parseInt(hideUntil, 10)) {
      setIsOpen(false);
    } else {
      setIsOpen(true);
    }

    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.key === 'Escape') {
        setIsOpen(false);
      }
    };
    
    if (isOpen) {
      window.addEventListener('keydown', handleKeyDown);
    }
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, [isOpen]);

  const closePopup = () => setIsOpen(false);

  const hideForToday = () => {
    const twentyFourHours = 24 * 60 * 60 * 1000;
    const hideUntil = new Date().getTime() + twentyFourHours;
    localStorage.setItem('hidePopupUntil', hideUntil.toString());
    setIsOpen(false);
  };

  if (!isOpen) return null;

  return (
    <div className="fixed inset-0 z-[100] flex items-center justify-center bg-black/60 backdrop-blur-sm p-4 lg:p-6" onClick={closePopup}>
      <motion.div 
        initial={{ opacity: 0, scale: 0.95 }}
        animate={{ opacity: 1, scale: 1 }}
        exit={{ opacity: 0, scale: 0.95 }}
        className="relative bg-white rounded-2xl shadow-2xl flex flex-col w-full max-w-[520px] max-h-[90vh] overflow-hidden"
        onClick={(e) => e.stopPropagation()}
      >
        <button 
          onClick={closePopup}
          className="absolute top-3 right-3 lg:top-4 lg:right-4 z-10 w-8 h-8 flex items-center justify-center bg-black/40 hover:bg-black/70 text-white rounded-full transition-colors backdrop-blur-md"
          aria-label="닫기"
        >
          <X className="w-5 h-5" />
        </button>
        <div className="overflow-y-auto w-full flex-grow flex items-center justify-center bg-[#171923]">
          <img 
            src="https://res.cloudinary.com/dxjz9ksjg/image/upload/v1787966113/ChatGPT_Image_2026%EB%85%84_8%EC%9B%94_29%EC%9D%BC_%EC%98%A4%EC%A0%84_10_10_01_kt4rmd.png" 
            alt="안내 팝업" 
            className="w-full h-auto object-contain block m-0 p-0"
          />
        </div>
        <div className="bg-white border-t border-slate-100 flex items-center justify-between px-5 py-4 shrink-0">
          <label className="flex items-center gap-2 cursor-pointer text-slate-600 hover:text-slate-900 transition-colors">
            <input 
              type="checkbox" 
              className="w-4 h-4 rounded border-slate-300 text-indigo-600 focus:ring-indigo-600 cursor-pointer"
              onChange={hideForToday}
            />
            <span className="text-[13px] lg:text-sm font-medium">오늘 하루 보지 않기</span>
          </label>
          <button 
            onClick={closePopup}
            className="text-[13px] lg:text-sm font-medium text-slate-500 hover:text-slate-900 transition-colors"
          >
            닫기
          </button>
        </div>
      </motion.div>
    </div>
  );
};
